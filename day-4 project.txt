cat > script.sh

#!/bin/bash
## Worker Node Script
echo "jenkins worker installation starts ..."
sleep 2
sudo apt update && sudo apt upgrade -y && sudo apt autoremove -y
sleep 2

## For JDK installation
sudo apt install openjdk-21-jdk -y
echo "JDK installation success"
sleep 2

## For Maven installation
sudo apt install maven -y
echo "maven installation success"
sleep 2

## For Git installation
sudo apt install git -y
echo "git installation success"
sleep 2

## For Docker installation
# Add Docker's official GPG key:
sudo apt update
sleep 2
sudo apt install ca-certificates curl -y
sleep 2
sudo install -m 0755 -d /etc/apt/keyrings
sleep 2
sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc
sleep 2
sudo chmod a+r /etc/apt/keyrings/docker.asc
sleep 2

# Add the repository to Apt sources:
sudo tee /etc/apt/sources.list.d/docker.sources <<EOF
Types: deb
URIs: https://download.docker.com/linux/ubuntu
Suites: $(. /etc/os-release && echo "$VERSION_CODENAME")
Components: stable
Architectures: $(dpkg --print-architecture)
Signed-By: /etc/apt/keyrings/docker.asc
EOF
sleep 2
sudo apt update
sleep 2
sudo apt install docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin -y
sleep 2
sudo systemctl start docker
sleep 2
sudo systemctl enable docker
sleep 2
sudo usermod -a -G docker ubuntu
sleep 2
echo -e "\n Docker success \n"

## For Trivy
echo "trivy installation starts ..."
sleep 2
sudo apt update && sudo apt upgrade -y
sleep 2
sudo apt-get install wget gnupg
sleep 2
wget -qO - https://aquasecurity.github.io/trivy-repo/deb/public.key | gpg --dearmor | sudo tee /usr/share/keyrings/trivy.gpg > /dev/null
sleep 2
echo "deb [signed-by=/usr/share/keyrings/trivy.gpg] [https://aquasecurity.github.io/trivy-repo/deb](https://aquasecurity.github.io/trivy-repo/deb) generic main" | sudo tee -a /etc/apt/sources.list.d/trivy.list
sleep 2
sudo apt-get update
sleep 2
sudo apt-get install trivy -y
trivy --version
# sudo mkdir -p /data/trivy-cache
# sudo chown -R $(whoami):$(whoami) /data/trivy-cache
echo "trivy installation success"

## For AWS CLI
sudo apt update
sudo apt install -y unzip curl
curl "https://awscli.amazonaws.com/awscli-exe-linux-x86_64.zip" -o "awscliv2.zip"
unzip awscliv2.zip
sudo ./aws/install
aws --version
echo "AWS CLI installation success"
echo -e "\n Installation success, logout and login again. \n"
