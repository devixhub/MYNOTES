name: Shell Script Workflow

on:
  push:

jobs:
  job-1:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Give Execute Permission
        run: chmod +x scripts/deploy.sh

      - name: Execute Shell Script
        run: ./scripts/deploy.sh

  job-2:
    runs-on: ubuntu-latest
    needs: job-1

    steps:
      - name: Final Step
        run: echo "Job 2 executed after Job 1"

  job-3:
    runs-on: ubuntu-latest
    needs: job-2

    steps:
      - name: Test Apache
        run: |
          sudo systemctl start apache2
          curl http://localhost

name: Maven Build and Docker Push
on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout source code
        uses: actions/checkout@v4
      - name: Setup Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
          cache: maven
      - name: Check Java Version
        run: java -version
      - name: Check Maven Version
        run: mvn -version
      - name: Maven Clean
        run: mvn clean
      - name: Maven Test
        run: mvn test
      - name: Maven Package
        run: mvn package -DskipTests
      - name: Check JAR File
        run: ls -lh target/

- name: Login to Docker Hub

          username: ${{ secrets.DOCKERHUB_USERNAME }}
          password: ${{ secrets.DOCKERHUB_TOKEN }}
      - name: Build Docker Image
        if: github.event_name == 'push'
        run: |
          docker build \
          -t ${{ secrets.DOCKERHUB_USERNAME }}/maven-demo:${{ github.run_number }} \
          -t ${{ secrets.DOCKERHUB_USERNAME }}/maven-demo:latest \
          .
      - name: Check Docker Images
        if: github.event_name == 'push'
        run: docker images
      - name: Push Version Image
        if: github.event_name == 'push'
        run: |
          docker push \
          ${{ secrets.DOCKERHUB_USERNAME }}/maven-demo:${{ github.run_number }}
      - name: Push Latest Image
        if: github.event_name == 'push'
        run: |
          docker push \
          ${{ secrets.DOCKERHUB_USERNAME }}/maven-demo:latest

name: Terraform Deployment

on: push

jobs:
  deploy:
    env:
      AWS_ACCESS_KEY_ID: "${{ secrets.AWS_ACCESS_KEY_ID }}"
      AWS_SECRET_ACCESS_KEY: "${{ secrets.AWS_SECRET_ACCESS_KEY }}"

    runs-on: ubuntu-latest

    steps:
      - name: Code Checkout
        uses: actions/checkout@v4

      - name: Set Up Terraform
        uses: hashicorp/setup-terraform@v3
        with:
          terraform_version: 1.12.2
      - name: Terraform Init, Plan
         run: |
          terraform init
          terraform plan

      - name: Terraform Apply
         run: terraform apply -auto-approve
